import cv2
import numpy as np
import matplotlib.pyplot as plt

image = cv2.imread('selfie.jpg', 0)

noise = np.random.normal(110, 5, image.shape)
noisy_image = image + noise

kernel = np.array([
    [1/9,1/9,1/9],
    [1/9,1/9,1/9],
    [1/9,1/9,1/9]
], dtype=np.float32)

recovered_image = cv2.filter2D(noisy_image, -1, kernel)

mse=np.mean((image - recovered_image)**2)

plt.figure(figsize=(10,10))

# CODE CELL 9

import cv2
import numpy as np
import matplotlib.pyplot as plt

img = cv2.imread('qn11.jpg', 0)
h, w = img.shape

R = np.array([[0, -1], [1, 0]])       # Rotation: 90 degrees
S = np.array([[2, 0], [0, 2]])        # Scaling
H = np.array([[1, 0.5], [0, 1]])      # Shearing

def transform(A):
    out = np.zeros_like(img)
    inv = np.linalg.inv(A)

    for y in range(h):
        for x in range(w):
            p = np.array([x-w//2, y-h//2])
            q = np.round(inv @ p + [w//2, h//2]).astype(int)
            u, v = q

            if 0 <= u < w and 0 <= v < h:
                out[y, x] = img[v, u]

    return out

plt.figure(figsize=(10, 7))

plt.subplot(2, 2, 1)
plt.imshow(img, cmap='gray')
plt.title('Original')
plt.axis('off')

plt.subplot(2, 2, 2)
plt.imshow(transform(R), cmap='gray')
plt.title('Rotation')
plt.axis('off')

plt.subplot(2, 2, 3)
plt.imshow(transform(S), cmap='gray')
plt.title('Scaling')
plt.axis('off')

plt.subplot(2, 2, 4)
plt.imshow(transform(H), cmap='gray')
plt.title('Shearing')
plt.axis('off')

plt.tight_layout()
plt.show()

# CODE CELL 10

import cv2
import numpy as np
import matplotlib.pyplot as plt

# Read video and track leaf apex
cap = cv2.VideoCapture('qn12.mp4')
rows, cols = [], []

while True:
    ret, frame = cap.read()
    if not ret:
        break

    plt.figure()
    plt.imshow(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
    plt.title('Leaf Frame')
    plt.axis('off')
    plt.show()

    r = int(input("Enter apex row: "))
    c = int(input("Enter apex column: "))
    rows.append(r)
    cols.append(c)

    if input("Continue? (y/n): ").lower() != 'y':
        break

cap.release()

# 1D FFT of row and column coordinates
rows = np.array(rows)
cols = np.array(cols)

R = np.fft.fft(rows - np.mean(rows))
C = np.fft.fft(cols - np.mean(cols))

# Read a frame for 2D FFT
img = cv2.imread('leaf_frame.jpg', 0)
F = np.fft.fft2(img)
Fshift = np.fft.fftshift(F)
spectrum = np.log(1 + np.abs(Fshift))

# Display all results
plt.figure(figsize=(10, 8))

plt.subplot(2, 2, 1)
plt.plot(np.abs(R))
plt.title('1D FFT - Row')
plt.xlabel('Frequency')

plt.subplot(2, 2, 2)
plt.plot(np.abs(C))
plt.title('1D FFT - Column')
plt.xlabel('Frequency')

plt.subplot(2, 2, 3)
plt.imshow(img, cmap='gray')
plt.title('Original Image')
plt.axis('off')

plt.subplot(2, 2, 4)
plt.imshow(spectrum, cmap='gray')
plt.title('2D FFT Spectrum')
plt.axis('off')

plt.tight_layout()
plt.show()
